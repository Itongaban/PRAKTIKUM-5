### Pseudocode
BEGIN

    INPUT a
    INPUT b

    penjumlahan ← a + b
    pengurangan ← a - b
    perkalian ← a × b
    pembagianInteger ← a / b
    pembagianReal ← a × 1.0 / b
    sisa ← a % b

    OUTPUT "a + b = " + penjumlahan
    OUTPUT "a - b = " + pengurangan
    OUTPUT "a * b = " + perkalian
    OUTPUT "a / b = " + pembagianInteger
    OUTPUT "a * 1.0 / b = " + pembagianReal
    OUTPUT "a % b = " + sisa

END
